set(TARGET llama-speculative)
add_executable(${TARGET} speculative.cpp)
install(TARGETS ${TARGET} RUNTIME)
target_link_libraries(${TARGET} PRIVATE common llama llama-hybrid-driver ${CMAKE_THREAD_LIBS_INIT})
target_compile_features(${TARGET} PRIVATE cxx_std_17)

find_path(NVTX_INCLUDE_DIR nvToolsExt.h PATHS /usr/local/cuda/include)
find_path(NVTX3_INCLUDE_DIR nvtx3/nvToolsExt.h PATHS /usr/local/cuda/include)
find_library(NVTX_LIBRARY nvToolsExt PATHS /usr/local/cuda/lib64)
if (NVTX3_INCLUDE_DIR)
    target_include_directories(${TARGET} PRIVATE ${NVTX3_INCLUDE_DIR})
    target_compile_definitions(${TARGET} PRIVATE LLAMA_SPEC_NVTX3=1)
    target_link_libraries(${TARGET} PRIVATE ${CMAKE_DL_LIBS})
elseif (NVTX_INCLUDE_DIR AND NVTX_LIBRARY)
    target_include_directories(${TARGET} PRIVATE ${NVTX_INCLUDE_DIR})
    target_compile_definitions(${TARGET} PRIVATE LLAMA_SPEC_NVTX2=1)
    target_link_libraries(${TARGET} PRIVATE ${NVTX_LIBRARY})
endif()
